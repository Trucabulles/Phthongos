cmake_minimum_required(VERSION 3.24)
project(PhthongosJUCE VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
include(cmake/JuceInputQueues.cmake)
add_subdirectory(vendor/JUCE)
add_subdirectory(core)
include(cmake/Translations.cmake)
juce_add_gui_app(PhthongosJUCE PRODUCT_NAME "Phthongos" COMPANY_NAME "Phthongos" ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/assets/branding/phthongos-icon.png" NEEDS_WEB_BROWSER FALSE NEEDS_CURL FALSE)
target_sources(PhthongosJUCE PRIVATE native/Main.cpp native/BlankProject.cpp native/MidiEntry.cpp native/Metronome.cpp native/StackedVoices.cpp native/StackedTests.cpp native/DialogOwner.cpp native/PointerSafety.cpp native/App.cpp native/Files.cpp native/Settings.cpp native/Analysis.cpp native/Tests.cpp native/MidiAddition.cpp native/SpectralOptions.cpp native/Guided.cpp native/Localisation.cpp native/LocalisationTests.cpp native/InstanceTests.cpp)
target_compile_options(PhthongosJUCE PRIVATE /utf-8)
target_sources(PhthongosJUCE PRIVATE native/ProjectCompatibilityTests.cpp)
target_sources(PhthongosJUCE PRIVATE native/FileTransfers.cpp native/WindowsFileDrag.cpp native/ProjectName.cpp native/Transcription.cpp native/BasicPitchTests.cpp native/OptionalResources.cpp native/RenderTests.cpp)
target_compile_definitions(PhthongosJUCE PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 PHT_WINDOWS_ISOLATE_INPUT_QUEUES=1)
target_link_libraries(PhthongosJUCE PRIVATE pht_services juce::juce_gui_basics ws2_32 userenv bcrypt ntdll crypt32 secur32 advapi32 ole32 propsys shell32 uuid)


target_include_directories(PhthongosJUCE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

option(PHT_BUILD_TEMPO_CHECKS "Build standalone C++ MIDI tempo reference checks" OFF)
if(PHT_BUILD_TEMPO_CHECKS)
  add_executable(pht_tempo_port_checks migration/TempoPortChecks.cpp vendor/midi-bpm/Detector.cpp)
endif()
